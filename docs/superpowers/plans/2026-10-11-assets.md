## Review Focus

1. **One source failing.** Binance down, or the key revoked, must not hide gold, and a price-source failure for one class must not hide the other. Expect the last cached values or a per-source error, never a blank page (Task 4 test).
2. **Coins Binance can't price.** For example `NFT`, or a delisted coin with no `<X>USDT` pair: listed with value null, excluded from totals and the chart, never NaN (Tasks 3 and 4 tests).
3. **Selling more gold than owned, or all of it.** If quantity reaches 0 or goes negative, the holding disappears or shows 0 without a divide-by-zero in the average cost (Task 4 test).
4. **A saved key whose secret is wrong** (Binance answers 401 or a -2014/-1022 code): `PUT /sources/binance` answers 400 with Binance's message, stores nothing, and never echoes the secret (Task 5 test).
5. **Switching KRW↔IDR with costs in other currencies.** A cost entered in USD for BTC and gold paid in IDR must convert correctly in both display currencies (Task 4 test).

```ts
import crypto from 'crypto';
import config from 'config';

// AES-256-GCM for secrets at rest (the Binance API secret). The key lives only in the server env.
export type Sealed = { secretEnc: string; iv: string; authTag: string };

const key = () => {
  const raw = config.get<string>('assetsEncryptionKey');
  const buf = Buffer.from(raw || '', 'base64');
  if (buf.length !== 32) throw new Error('ASSETS_ENCRYPTION_KEY must be 32 bytes, base64');
  return buf;
};

export const isEncryptionConfigured = () => {
  try {
    key();
    return true;
  } catch {
    return false;
  }
};

export function encrypt(plain: string): Sealed {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv('aes-256-gcm', key(), iv);
  const enc = Buffer.concat([cipher.update(plain, 'utf8'), cipher.final()]);
  return { secretEnc: enc.toString('base64'), iv: iv.toString('base64'), authTag: cipher.getAuthTag().toString('base64') };
}

export function decrypt(s: Sealed): string {
  const decipher = crypto.createDecipheriv('aes-256-gcm', key(), Buffer.from(s.iv, 'base64'));
  decipher.setAuthTag(Buffer.from(s.authTag, 'base64'));
  return Buffer.concat([decipher.update(Buffer.from(s.secretEnc, 'base64')), decipher.final()]).toString('utf8');
}
```

```ts
import crypto from 'crypto';

// The only caller of signed Binance endpoints. Read-only: account data, never orders or transfers.
export const BINANCE_BASE = 'https://api.binance.com';
export type BinanceCreds = { apiKey: string; apiSecret: string };
export class BinanceError extends Error {}

const FORBIDDEN = [
  'enableSpotAndMarginTrading', 'enableWithdrawals', 'enableInternalTransfer',
  'enableMargin', 'enableFutures', 'permitsUniversalTransfer', 'enableVanillaOptions',
] as const;

export const sign = (query: string, secret: string) => crypto.createHmac('sha256', secret).update(query).digest('hex');

async function signedRequest<T>(method: 'GET' | 'POST', path: string, params: Record<string, string | number>, creds: BinanceCreds): Promise<T> {
  const query = new URLSearchParams({ ...Object.fromEntries(Object.entries(params).map(([k, v]) => [k, String(v)])), recvWindow: '10000', timestamp: String(Date.now()) }).toString();
  const url = `${BINANCE_BASE}${path}?${query}&signature=${sign(query, creds.apiSecret)}`;
  let res: Response;
  try {
    res = await fetch(url, { method, headers: { 'X-MBX-APIKEY': creds.apiKey }, signal: AbortSignal.timeout(10_000) });
  } catch {
    console.error(`Binance request failed: ${path} network`);
    throw new BinanceError("Couldn't reach Binance");
  }
  const body = (await res.json().catch(() => ({}))) as { msg?: string };
  if (!res.ok) {
    const msg = body.msg ?? `HTTP ${res.status}`;
    console.error(`Binance request failed: ${path} ${res.status} ${msg}`);
    throw new BinanceError(msg);
  }
  return body as T;
}

export async function checkReadOnly(creds: BinanceCreds): Promise<void> {
  const r = await signedRequest<Record<string, boolean>>('GET', '/sapi/v1/account/apiRestrictions', {}, creds);
  if (!r.enableReading || FORBIDDEN.some((p) => r[p])) {
    throw new BinanceError('Use a read-only key: turn off trading, withdrawals and transfers');
  }
}

// Spot + Funding + Simple Earn (flexible and locked), summed per coin
export async function fetchBalances(creds: BinanceCreds): Promise<Record<string, number>> {
  const add = (acc: Record<string, number>, asset: string, amount: number) => {
    if (!amount || asset.startsWith('LD')) return; // LD* = legacy savings tokens, mirrored by Earn
    acc[asset] = (acc[asset] ?? 0) + amount;
  };
  const [spot, funding, flexible, locked] = await Promise.all([
    signedRequest<{ balances: { asset: string; free: string; locked: string }[] }>('GET', '/api/v3/account', { omitZeroBalances: 'true' }, creds),
    signedRequest<{ asset: string; free: string; locked: string; freeze: string }[]>('POST', '/sapi/v1/asset/get-funding-asset', {}, creds),
    signedRequest<{ rows: { asset: string; totalAmount: string }[] }>('GET', '/sapi/v1/simple-earn/flexible/position', { size: 100 }, creds),
    signedRequest<{ rows: { asset: string; amount: string }[] }>('GET', '/sapi/v1/simple-earn/locked/position', { size: 100 }, creds),
  ]);
  const acc: Record<string, number> = {};
  for (const b of spot.balances) add(acc, b.asset, Number(b.free) + Number(b.locked));
  for (const b of funding) add(acc, b.asset, Number(b.free) + Number(b.locked) + Number(b.freeze));
  for (const r of flexible.rows ?? []) add(acc, r.asset, Number(r.totalAmount));
  for (const r of locked.rows ?? []) add(acc, r.asset, Number(r.amount));
  return acc;
}
```

```ts
import { BINANCE_BASE } from '../binance.service';
import type { AssetClass } from './asset-classes';

export const GRAMS_PER_TROY_OZ = 31.1035;
const PRICE_TTL = 60_000;
const FX_TTL = 3_600_000;
export class PriceUnavailableError extends Error {}

type Entry<T> = { at: number; value: T };
const priceCache = new Map<string, Entry<number | null>>(); // key: CLASS:SYMBOL
let fxCache: Entry<{ KRW: number; IDR: number }> | null = null;
export const resetPriceCache = () => { priceCache.clear(); fxCache = null; };

const pair = (cls: AssetClass, symbol: string) => (cls === 'GOLD' ? 'PAXGUSDT' : `${symbol}USDT`);

async function tickers(pairs: string[]): Promise<Record<string, number>> {
  const get = async (q: string) => {
    const res = await fetch(`${BINANCE_BASE}/api/v3/ticker/price?${q}`, { signal: AbortSignal.timeout(10_000) });
    if (!res.ok) throw Object.assign(new Error('ticker'), { status: res.status });
    return res.json() as Promise<{ symbol: string; price: string } | { symbol: string; price: string }[]>;
  };
  const toMap = (rows: { symbol: string; price: string }[]) => Object.fromEntries(rows.map((r) => [r.symbol, Number(r.price)]));
  try {
    const rows = await get(`symbols=${encodeURIComponent(JSON.stringify(pairs))}`);
    return toMap(Array.isArray(rows) ? rows : [rows]);
  } catch (err) {
    // One unknown pair fails the whole batch (400): fall back to one request per pair
    if ((err as { status?: number }).status !== 400) throw err;
    const out: Record<string, number> = {};
    await Promise.all(pairs.map(async (p) => {
      try {
        const row = (await get(`symbol=${p}`)) as { symbol: string; price: string };
        out[p] = Number(row.price);
      } catch { /* no such pair */ }
    }));
    return out;
  }
}

// USD price per unit (gram for GOLD, coin for CRYPTO); null when Binance has no price
export async function usdPrices(cls: AssetClass, symbols: string[]) {
  const now = Date.now();
  const prices: Record<string, number | null> = {};
  const need: string[] = [];
  for (const s of symbols) {
    if (cls === 'CRYPTO' && s === 'USDT') { prices[s] = 1; continue; }
    const hit = priceCache.get(`${cls}:${s}`);
    if (hit && now - hit.at < PRICE_TTL) prices[s] = hit.value;
    else need.push(s);
  }
  if (need.length === 0) return { prices, stale: false };
  try {
    const got = await tickers([...new Set(need.map((s) => pair(cls, s)))]);
    for (const s of need) {
      const raw = got[pair(cls, s)];
      const value = raw === undefined ? null : cls === 'GOLD' ? raw / GRAMS_PER_TROY_OZ : raw;
      priceCache.set(`${cls}:${s}`, { at: now, value });
      prices[s] = value;
    }
    return { prices, stale: false };
  } catch {
    const missing = need.filter((s) => !priceCache.has(`${cls}:${s}`));
    if (missing.length) throw new PriceUnavailableError('Prices are unavailable right now');
    for (const s of need) prices[s] = priceCache.get(`${cls}:${s}`)!.value;
    return { prices, stale: true };
  }
}

export async function usdRates() {
  const now = Date.now();
  if (fxCache && now - fxCache.at < FX_TTL) return { rates: fxCache.value, stale: false };
  try {
    const res = await fetch('https://open.er-api.com/v6/latest/USD', { signal: AbortSignal.timeout(10_000) });
    const body = (await res.json()) as { result: string; rates: Record<string, number> };
    if (body.result !== 'success') throw new Error('fx');
    fxCache = { at: now, value: { KRW: body.rates.KRW, IDR: body.rates.IDR } };
    return { rates: fxCache.value, stale: false };
  } catch {
    if (!fxCache) throw new PriceUnavailableError('Prices are unavailable right now');
    return { rates: fxCache.value, stale: true };
  }
}
```

```ts
import prisma from '../../middleware/prismaMiddleware';
import { decrypt } from '../crypto-box';
import { fetchBalances, BinanceError } from '../binance.service';
import type { AssetClass } from './asset-classes';

export type Holding = { assetClass: AssetClass; symbol: string; quantity: number; source: 'manual' | 'binance' };
export type BinanceStatus = { connected: boolean; lastSyncedAt: string | null; error: string | null };

// Manual lots summed per (class, symbol); fully sold holdings drop out
export async function manualHoldings(userId: string): Promise<Holding[]> {
  const lots = await prisma.assetLot.findMany({ where: { userId } });
  const sums = new Map<string, number>();
  for (const l of lots) {
    const key = `${l.assetClass}:${l.symbol}`;
    sums.set(key, (sums.get(key) ?? 0) + l.quantity.toNumber());
  }
  return [...sums].filter(([, q]) => q > 1e-9).map(([key, quantity]) => {
    const [assetClass, symbol] = key.split(':');
    return { assetClass: assetClass as AssetClass, symbol, quantity, source: 'manual' as const };
  });
}

const BINANCE_TTL = 5 * 60_000;
const binanceCache = new Map<string, { at: number; balances: Record<string, number> }>();
export const resetBinanceCache = () => binanceCache.clear();

export async function binanceHoldings(userId: string, refresh: boolean): Promise<{ holdings: Holding[]; status: BinanceStatus }> {
  const conn = await prisma.binanceConnection.findUnique({ where: { userId } });
  if (!conn) return { holdings: [], status: { connected: false, lastSyncedAt: null, error: null } };

  const cached = binanceCache.get(userId);
  let balances = cached?.balances ?? {};
  let lastSyncedAt = conn.lastSyncedAt;
  let error = conn.lastError;
  if (refresh || !cached || Date.now() - cached.at >= BINANCE_TTL) {
    try {
      balances = await fetchBalances({ apiKey: conn.apiKey, apiSecret: decrypt(conn) });
      binanceCache.set(userId, { at: Date.now(), balances });
      lastSyncedAt = new Date();
      error = null;
      await prisma.binanceConnection.update({ where: { userId }, data: { lastSyncedAt, lastError: null } });
    } catch (err) {
      error = err instanceof BinanceError ? err.message : "Couldn't reach Binance";
      await prisma.binanceConnection.update({ where: { userId }, data: { lastError: error } });
    }
  }
  const holdings = Object.entries(balances)
    .filter(([, q]) => q > 0)
    .map(([symbol, quantity]) => ({ assetClass: 'CRYPTO' as const, symbol, quantity, source: 'binance' as const }));
  return { holdings, status: { connected: true, lastSyncedAt: lastSyncedAt?.toISOString() ?? null, error } };
}
```

```ts
import prisma from '../../middleware/prismaMiddleware';
import { ASSET_CLASSES, type AssetClass } from './asset-classes';
import { usdPrices, usdRates, PriceUnavailableError } from './prices';
import { manualHoldings, binanceHoldings, type Holding } from './sources';

export const DUST_USD = 1;
type Currency = 'KRW' | 'IDR';
const toUsd = (amount: number, currency: string, rates: { KRW: number; IDR: number }) =>
  currency === 'USD' ? amount : amount / rates[currency as Currency];

export async function getPortfolio(userId: string, currency: Currency, refresh: boolean) {
  const [{ rates, stale: fxStale }, manual, binance, lots, costs] = await Promise.all([
    usdRates(), // throws PriceUnavailableError when there's no rate at all: the route answers 503
    manualHoldings(userId),
    binanceHoldings(userId, refresh),
    prisma.assetLot.findMany({ where: { userId } }),
    prisma.assetCost.findMany({ where: { userId } }),
  ]);
  const holdings: Holding[] = [...manual, ...binance.holdings];

  // Price each class separately so one failing source doesn't hide the others
  let stale = fxStale;
  const usd: Record<string, number | null> = {};
  for (const cls of Object.keys(ASSET_CLASSES) as AssetClass[]) {
    const symbols = [...new Set(holdings.filter((h) => h.assetClass === cls).map((h) => h.symbol))];
    if (symbols.length === 0) continue;
    try {
      const r = await usdPrices(cls, symbols);
      stale ||= r.stale;
      for (const s of symbols) usd[`${cls}:${s}`] = r.prices[s];
    } catch (err) {
      if (!(err instanceof PriceUnavailableError)) throw err;
      for (const s of symbols) usd[`${cls}:${s}`] = null;
    }
  }

  const rate = rates[currency];
  const costUsd = (h: Holding): number | null => {
    if (h.source === 'manual') {
      return lots
        .filter((l) => l.assetClass === h.assetClass && l.symbol === h.symbol)
        .reduce((sum, l) => sum + toUsd(l.totalPaid.toNumber(), l.currency, rates), 0);
    }
    const c = costs.find((c) => c.assetClass === h.assetClass && c.symbol === h.symbol);
    return c ? toUsd(c.costAmount.toNumber(), c.costCurrency, rates) : null;
  };

  const rows = holdings
    .map((h) => {
      const price = usd[`${h.assetClass}:${h.symbol}`] ?? null;
      const valueUsd = price === null ? null : h.quantity * price;
      const cUsd = costUsd(h);
      const value = valueUsd === null ? null : valueUsd * rate;
      const cost = cUsd === null ? null : cUsd * rate;
      const gain = value !== null && cost !== null ? value - cost : null;
      return {
        assetClass: h.assetClass, symbol: h.symbol, quantity: h.quantity,
        unit: ASSET_CLASSES[h.assetClass].unit, source: h.source,
        value, cost, gain, gainPct: gain !== null && cost ? (gain / cost) * 100 : null,
        valueUsd,
      };
    })
    .filter((r) => r.valueUsd === null || r.valueUsd >= DUST_USD)
    .sort((a, b) => (b.value ?? -1) - (a.value ?? -1))
    .map(({ valueUsd: _drop, ...r }) => r);

  const priced = rows.filter((r) => r.value !== null);
  const withCost = priced.filter((r) => r.cost !== null);
  const totalValue = priced.reduce((s, r) => s + r.value!, 0);
  const totalCost = withCost.length ? withCost.reduce((s, r) => s + r.cost!, 0) : null;
  const costedValue = withCost.reduce((s, r) => s + r.value!, 0);
  const totalGain = totalCost === null ? null : costedValue - totalCost;

  return {
    currency,
    total: { value: totalValue, cost: totalCost, gain: totalGain, gainPct: totalGain !== null && totalCost ? (totalGain / totalCost) * 100 : null },
    holdings: rows,
    sources: { binance: binance.status },
    stale,
  };
}
```

`client/src/components/assets/distribution.ts`:

```ts
import { ASSET_CLASS_UI, holdingLabel } from './asset-classes';

const PALETTE = ['#F7931A', '#627EEA', '#26A17B', '#8247E5', '#E84142', '#00AAE4', '#9CA3AF'];
const OTHER = '#9CA3AF';

// Donut slices: each priced holding's share; slices under `threshold`% merge into "Other"
export function toSlices(holdings: { assetClass: string; symbol: string; value: number | null }[], threshold = 3) {
  const priced = holdings.filter((h) => h.value !== null && h.value > 0) as { assetClass: string; symbol: string; value: number }[];
  const total = priced.reduce((s, h) => s + h.value, 0);
  if (total <= 0) return [];
  const sorted = [...priced].sort((a, b) => b.value - a.value);
  const big = sorted.filter((h) => (h.value / total) * 100 >= threshold);
  const small = sorted.filter((h) => (h.value / total) * 100 < threshold);
  let p = 0;
  const slices = big.map((h) => ({
    name: holdingLabel(h),
    value: h.value,
    pct: (h.value / total) * 100,
    color: ASSET_CLASS_UI[h.assetClass]?.color ?? PALETTE[p++ % PALETTE.length],
  }));
  const otherValue = small.reduce((s, h) => s + h.value, 0);
  if (otherValue > 0) slices.push({ name: 'Other', value: otherValue, pct: (otherValue / total) * 100, color: OTHER });
  return slices;
}
```
